15/**
16 * @brief Computes the next iteration for a single point in the Mandelbrot set.
17 */
18static void compute_next_iteration(t_render_vars *vars, int i)
19{
20 double temp_z_re;
21
22 temp_z_re = vars->z_squared[COMPLEX_RE] - vars->z_squared[COMPLEX_IM];
23 temp_z_re += vars->c_re[i];
24 vars->z[COMPLEX_IM] = (2.0 * vars->z[COMPLEX_RE] * vars->z[COMPLEX_IM]);
25 vars->z[COMPLEX_IM] += vars->c_im;
26 vars->z[COMPLEX_RE] = temp_z_re;
27 vars->z_squared[COMPLEX_RE] = vars->z[COMPLEX_RE] * vars->z[COMPLEX_RE];
28 vars->z_squared[COMPLEX_IM] = vars->z[COMPLEX_IM] * vars->z[COMPLEX_IM];
29}

31/**
32 * @brief Performs the iterations for a single point in the Mandelbrot set.
33 */
34static void mandelbrot_iterations(t_render_vars *vars, int i)
35{
36 double current_magnitude_squared;
37 double escape_radius_squared;
38
39 escape_radius_squared = 4.0;
40 vars->z[COMPLEX_RE] = 0.0;
41 vars->z[COMPLEX_IM] = 0.0;
42 vars->z_squared[COMPLEX_RE] = 0;
43 vars->z_squared[COMPLEX_IM] = 0;
44 vars->iterations = 0;
45 while (1)
46 {
47 current_magnitude_squared = vars->z_squared[COMPLEX_RE];
48 current_magnitude_squared += vars->z_squared[COMPLEX_IM];
49 if (vars->iterations >= MAX_ITERATIONS)
50 break ;
51 if (current_magnitude_squared > escape_radius_squared)
52 break ;
53 compute_next_iteration(vars, i);
54 vars->iterations++;
55 }
56}

58/**
59 * @brief Performs an early scape check for a point in the Mandelbrot set.
60 */
61static int mandelbrot_escape_check(t_render_vars *vars, int i, t_data *data)
62{
63 int max_color;
64 double dist_sq;
65 double esc_chk;
66 double bound_condition;
67
68 max_color = calculate_color(MAX_ITERATIONS);
69 vars->c_re_centered = vars->c_re[i] - 0.25;
70 dist_sq = vars->c_re_centered * vars->c_re_centered;
71 dist_sq += vars->c_im * vars->c_im;
72 esc_chk = dist_sq * (dist_sq + vars->c_re_centered);
73 if ((data->zoom > 1.0) && (esc_chk < 0.25 * vars->c_im * vars->c_im))
74 return (vars->row[i] = max_color, 1);
75 bound_condition = (vars->c_re[i] + 1) * (vars->c_re[i] + 1);
76 bound_condition += vars->c_im * vars->c_im;
77 if (bound_condition < 0.0625)
78 return (vars->row[i] = max_color, 1);
79 return (0);
80}

82/**
83 * @brief Renders a single row of the Mandelbrot set.
84 */
85static void render_mandelbrot_row(t_render_vars *vars, t_data *data, int y)
86{
87 int x;
88 int i;
89
90 x = 0;
91 while (x < WIDTH)
92 {
93 vars->row = vars->pixels + (y * WIDTH + x);
94 vars->c_re[0] = vars->start[COMPLEX_RE] + x * vars->scale[SCALE_X];
95 vars->c_re[1] = vars->start[COMPLEX_RE];
96 vars->c_re[1] += (x + 1) * vars->scale[SCALE_X];
97 i = 0;
98 while (i < 2)
99 {
100 if (mandelbrot_escape_check(vars, i, data))
101 {
102 i++;
103 continue ;
104 }
105 mandelbrot_iterations(vars, i);
106 vars->row[i++] = calculate_color(vars->iterations);
107 }
108 x += 2;
109 }
110}
